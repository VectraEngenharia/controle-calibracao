/** * SISTEMA DE CONTROLE DE CALIBRAÇÃO DE INSTRUMENTOS * Backend em Google Apps Script. * * Este script deve ser criado DENTRO da planilha Google Sheets * (Extensões > Apps Script), para que ele já saiba qual planilha usar. */ // ---------- CONFIGURAÇÃO ---------- const ID_PLANILHA = '1kBIjL40Nn8SatBRYe3UK589rYlvWkAgZ6c0s3c-LPdk'; const ID_PASTA_CERTIFICADOS = '1kwOHuf-8g3z0Lwy1Wkjlj6wVflRNDanO'; const ABA_INSTRUMENTOS = 'Instrumentos'; const ABA_ADMINS = 'Admins'; const ABA_HISTORICO = 'Historico'; // Colunas da aba Instrumentos, na ordem em que aparecem na planilha const COLUNAS_INSTRUMENTOS = [ 'ID', 'INSTRUMENTO', 'TAG', 'Nº DE SERIE', 'MODELO', 'MARCA', 'DATA DA AFERIÇÃO', 'DATA DO VENCIMENTO', 'LABORATÓRIO', 'CERTIFICADO', 'STATUS', 'INSPETOR', 'LINK_CERTIFICADO' ]; // ---------- PONTO DE ENTRADA (GET) ---------- // Usado para LEITURA pública: lista de instrumentos. function doGet(e) { const action = e.parameter.action || 'list'; if (action === 'list') { return jsonResponse({ ok: true, data: listarInstrumentos() }); } return jsonResponse({ ok: false, erro: 'Ação GET desconhecida.' }); } // ---------- PONTO DE ENTRADA (POST) ---------- // Usado para LOGIN e todas as ações de escrita (protegidas por login). function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonResponse({ ok: false, erro: 'Requisição inválida.' }); } const action = body.action; if (action === 'login') { const admin = validarLogin(body.usuario, body.senha); if (!admin) return jsonResponse({ ok: false, erro: 'Usuário ou senha inválidos.' }); return jsonResponse({ ok: true, nome: admin.nome }); } // Todas as ações abaixo exigem login válido em toda requisição const admin = validarLogin(body.usuario, body.senha); if (!admin) { return jsonResponse({ ok: false, erro: 'Sessão inválida. Faça login novamente.' }); } switch (action) { case 'addInstrument': return jsonResponse(adicionarInstrumento(body.instrumento, admin)); case 'removeInstrument': return jsonResponse(removerInstrumento(body.id, admin)); case 'reassignInspector': return jsonResponse(reatribuirInspetor(body.id, body.inspetor, admin)); case 'updateStatus': return jsonResponse(atualizarStatus(body.id, body.status, admin)); case 'updateCertLink': return jsonResponse(atualizarLinkCertificado(body.id, body.link, admin)); case 'uploadCertificado': return jsonResponse(uploadCertificado(body.id, body.conteudoBase64, admin)); case 'history': return jsonResponse({ ok: true, data: listarHistorico() }); default: return jsonResponse({ ok: false, erro: 'Ação POST desconhecida.' }); } } // ---------- LOGIN ---------- function validarLogin(usuario, senha) { if (!usuario || !senha) return null; const sheet = getSheet(ABA_ADMINS); const dados = sheet.getDataRange().getValues(); // Colunas: USUARIO, SENHA, NOME_EXIBICAO for (let i = 1; i < dados.length; i++) { const [u, s, nome] = dados[i]; if (String(u).trim() === String(usuario).trim() && String(s) === String(senha)) { return { usuario: u, nome: nome || u }; } } return null; } // ---------- LEITURA DE INSTRUMENTOS ---------- function listarInstrumentos() { const sheet = getSheet(ABA_INSTRUMENTOS); const dados = sheet.getDataRange().getValues(); const cabecalho = dados[0]; const linhas = dados.slice(1); return linhas .filter(l => l[0] !== '' && l[0] !== null) // ignora linhas vazias .map(l => { const obj = {}; cabecalho.forEach((col, i) => { let valor = l[i]; // Datas viram string ISO (AAAA-MM-DD) para o front-end tratar fácil if (valor instanceof Date) { valor = Utilities.formatDate(valor, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } obj[col] = valor; }); return obj; }); } // ---------- ADICIONAR INSTRUMENTO ---------- function adicionarInstrumento(instrumento, admin) { if (!instrumento || !instrumento.INSTRUMENTO) { return { ok: false, erro: 'Nome do instrumento é obrigatório.' }; } const sheet = getSheet(ABA_INSTRUMENTOS); const dados = sheet.getDataRange().getValues(); // Gera novo ID = maior ID atual + 1 let maiorId = 0; for (let i = 1; i < dados.length; i++) { const id = Number(dados[i][0]); if (!isNaN(id) && id > maiorId) maiorId = id; } const novoId = maiorId + 1; const novaLinha = COLUNAS_INSTRUMENTOS.map(col => { if (col === 'ID') return novoId; return instrumento[col] !== undefined ? instrumento[col] : ''; }); sheet.appendRow(novaLinha); registrarHistorico(admin, 'ADICIONAR_INSTRUMENTO', novoId, `Instrumento "${instrumento.INSTRUMENTO}" (tag ${instrumento.TAG || 'N/C'}) adicionado.`); return { ok: true, id: novoId }; } // ---------- REMOVER INSTRUMENTO ---------- function removerInstrumento(id, admin) { const sheet = getSheet(ABA_INSTRUMENTOS); const linha = encontrarLinhaPorId(sheet, id); if (linha === -1) return { ok: false, erro: 'Instrumento não encontrado.' }; const nomeInstrumento = sheet.getRange(linha, 2).getValue(); sheet.deleteRow(linha); registrarHistorico(admin, 'REMOVER_INSTRUMENTO', id, `Instrumento "${nomeInstrumento}" (ID ${id}) removido.`); return { ok: true }; } // ---------- REATRIBUIR INSPETOR ---------- function reatribuirInspetor(id, inspetor, admin) { const sheet = getSheet(ABA_INSTRUMENTOS); const linha = encontrarLinhaPorId(sheet, id); if (linha === -1) return { ok: false, erro: 'Instrumento não encontrado.' }; const colInspetor = COLUNAS_INSTRUMENTOS.indexOf('INSPETOR') + 1; const nomeInstrumento = sheet.getRange(linha, 2).getValue(); const inspetorAnterior = sheet.getRange(linha, colInspetor).getValue(); sheet.getRange(linha, colInspetor).setValue(inspetor || ''); registrarHistorico(admin, 'REATRIBUIR_INSPETOR', id, `Instrumento "${nomeInstrumento}" (ID ${id}): inspetor alterado de "${inspetorAnterior || '(sem inspetor)'}" para "${inspetor || '(sem inspetor)'}".`); return { ok: true }; } // ---------- ATUALIZAR STATUS (ativo/inativo) ---------- function atualizarStatus(id, status, admin) { const sheet = getSheet(ABA_INSTRUMENTOS); const linha = encontrarLinhaPorId(sheet, id); if (linha === -1) return { ok: false, erro: 'Instrumento não encontrado.' }; const colStatus = COLUNAS_INSTRUMENTOS.indexOf('STATUS') + 1; const nomeInstrumento = sheet.getRange(linha, 2).getValue(); const statusAnterior = sheet.getRange(linha, colStatus).getValue(); sheet.getRange(linha, colStatus).setValue(status || ''); registrarHistorico(admin, 'ATUALIZAR_STATUS', id, `Instrumento "${nomeInstrumento}" (ID ${id}): status alterado de "${statusAnterior || '(vazio)'}" para "${status || '(vazio)'}".`); return { ok: true }; } // ---------- ATUALIZAR LINK DO CERTIFICADO (manual, opcional) ---------- function atualizarLinkCertificado(id, link, admin) { const sheet = getSheet(ABA_INSTRUMENTOS); const linha = encontrarLinhaPorId(sheet, id); if (linha === -1) return { ok: false, erro: 'Instrumento não encontrado.' }; const colLink = COLUNAS_INSTRUMENTOS.indexOf('LINK_CERTIFICADO') + 1; const nomeInstrumento = sheet.getRange(linha, 2).getValue(); sheet.getRange(linha, colLink).setValue(link || ''); registrarHistorico(admin, 'ATUALIZAR_LINK_CERTIFICADO', id, `Instrumento "${nomeInstrumento}" (ID ${id}): link do certificado ${link ? 'atualizado manualmente' : 'removido'}.`); return { ok: true }; } // ---------- UPLOAD DO PDF DO CERTIFICADO (via Drive) ---------- // Salva o PDF na pasta de certificados, nomeado como "{ID}.pdf", // deixa público para leitura, e grava o link automaticamente na planilha. function uploadCertificado(id, conteudoBase64, admin) { const sheet = getSheet(ABA_INSTRUMENTOS); const linha = encontrarLinhaPorId(sheet, id); if (linha === -1) return { ok: false, erro: 'Instrumento não encontrado.' }; if (!conteudoBase64) return { ok: false, erro: 'Nenhum arquivo recebido.' }; const nomeInstrumento = sheet.getRange(linha, 2).getValue(); const nomeArquivo = id + '.pdf'; const pasta = DriveApp.getFolderById(ID_PASTA_CERTIFICADOS); // Se já existir um PDF com esse ID, remove antes de subir o novo (substitui) const existentes = pasta.getFilesByName(nomeArquivo); while (existentes.hasNext()) { existentes.next().setTrashed(true); } const bytes = Utilities.base64Decode(conteudoBase64); const blob = Utilities.newBlob(bytes, 'application/pdf', nomeArquivo); const arquivo = pasta.createFile(blob); arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); const link = 'https://drive.google.com/file/d/' + arquivo.getId() + '/preview'; const colLink = COLUNAS_INSTRUMENTOS.indexOf('LINK_CERTIFICADO') + 1; sheet.getRange(linha, colLink).setValue(link); registrarHistorico(admin, 'UPLOAD_CERTIFICADO', id, `Instrumento "${nomeInstrumento}" (ID ${id}): PDF do certificado enviado.`); return { ok: true, link: link }; } // ---------- HISTÓRICO ---------- function registrarHistorico(admin, acao, instrumentoId, detalhes) { const sheet = getSheet(ABA_HISTORICO); const agora = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm'); sheet.appendRow([agora, admin.usuario, acao, instrumentoId, detalhes]); } function listarHistorico() { const sheet = getSheet(ABA_HISTORICO); const dados = sheet.getDataRange().getValues(); const linhas = dados.slice(1).filter(l => l[0] !== '' && l[0] !== null); // Mais recente primeiro return linhas.reverse().map(l => ({ data_hora: l[0], admin: l[1], acao: l[2], instrumento_id: l[3], detalhes: l[4] })); } // ---------- AUXILIARES ---------- function getSheet(nome) { const planilha = SpreadsheetApp.openById(ID_PLANILHA); const sheet = planilha.getSheetByName(nome); if (!sheet) throw new Error('Aba não encontrada: ' + nome); return sheet; } function encontrarLinhaPorId(sheet, id) { const dados = sheet.getDataRange().getValues(); for (let i = 1; i < dados.length; i++) { if (String(dados[i][0]) === String(id)) return i + 1; // +1 porque getRange é 1-indexado } return -1; } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
